clear all
format short

syms x1 x2;

f1 = x1^2 - x1*x2 + x2^2;

fx = inline(f1);
fobj = @(x) fx(x(:,1), x(:,2));

% Gradient
grad = gradient(f1);
G = inline(grad);
gradx = @(Y) G(Y(:,1), Y(:,2));

% Hessian
H1 = hessian(f1);
Hx = inline(H1);

% Steepest descent method
tol = 0.05;
x0 = [1, 1/2];
maxiter = 1000;

X = [];
iter = 0;

fprintf('Iteration\t x1\t\t x2\t\t f(x)\n');

while norm(gradx(x0)) > tol && iter < maxiter
    
    X = [X, x0];
    
    % 🔴 PRINT CURRENT VALUES
    fprintf('%d\t\t %f\t %f\t %f\n', ...
        iter, x0(1), x0(2), fobj(x0));
    
    H = Hx(x0);
    S = -gradx(x0);
    
    lambda = (S' * S) / (S' * H * S);
    fprintf('lambda = %f\n', lambda);
    Xnew = x0 + lambda .* S';
    x0 = Xnew;
    
    iter = iter + 1;
end

% Final output
fprintf('\nOptimal Solution X = [%f, %f]\n', x0(1), x0(2));
fprintf('Optimal value f(x) = %f\n', fobj(x0));
